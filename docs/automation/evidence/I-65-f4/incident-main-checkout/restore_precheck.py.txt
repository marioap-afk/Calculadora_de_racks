def git(repo, *a, binary=False):
    r = subprocess.run(["git", "--no-optional-locks", "-C", repo] + list(a), capture_output=True)
    if r.returncode != 0:
        raise SystemExit("git %s failed: %s" % (a, r.stderr.decode("utf-8", "replace")))
    return r.stdout if binary else r.stdout.decode("utf-8", "replace")


def h(b):
    return hashlib.sha256(b).hexdigest().upper()


res = {"ObtainedUtc": datetime.datetime.now(datetime.timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ"), "checks": {}}
cur = open(os.path.join(MAIN, "RackCad.sln"), "rb").read()
