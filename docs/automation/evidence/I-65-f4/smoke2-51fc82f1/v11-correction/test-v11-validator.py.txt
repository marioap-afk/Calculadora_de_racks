"""I-65 segundo smoke: pruebas positivas y negativas de V-11 (validador v2). Solo LEE el paquete real; trabaja sobre copias.

Uso: python -I test_v11_validator.py <validador-v2> <validador-v1> <expected-inputs.json> <raiz-nueva-de-pruebas> <salida.json>
Cada caso copia los tres archivos del paquete real a <raiz>/<caso>/..., reubica la carga util con la MISMA serializacion que el
producto (JSON -> Escape de literal de Python, RamBuildScriptTemplate.Escape) y aplica UNA mutacion. Ejecuta el validador v2 como
subproceso y lee el resultado de V-11. Positivos: V-11 PASS. Negativos: V-11 FAIL. N11 comprueba ademas que el script no se ejecuto.
"""

v2, v1, expected, root, out = sys.argv[1:6]
REAL = "D:\\r65-smoke\\51fc82f1\\run-R20261016T1918Z\\pkg-01"
NAMES = ["manifest.json", "rackcad-ram-build.py", "structural-loads.json"]
real_bytes = {n: open(os.path.join(REAL, n), "rb").read() for n in NAMES}
real_sha_before = {n: hashlib.sha256(b).hexdigest() for n, b in real_bytes.items()}
os.makedirs(root)  # nueva; falla si existe
os.makedirs(os.path.join(root, "reports"))
script0 = real_bytes["rackcad-ram-build.py"].decode("utf-8")
manifest0 = real_bytes["manifest.json"].decode("utf-8")
lines0 = script0.split("\n")
PREFIX = 'PAYLOAD = "'
idx = [i for i, line in enumerate(lines0) if line.startswith(PREFIX)]
assert len(idx) == 1, idx
IDX = idx[0]
json0 = ast.literal_eval(lines0[IDX][len("PAYLOAD = "):])  # texto JSON de la carga util del paquete real
folder0 = json.loads(json0)["packageFolder"]
assert folder0 == REAL, folder0
fp0 = json.loads(json0)["analysisInputFingerprint"]


def py_escape(text):
    """Igual que RamBuildScriptTemplate.Escape: primero la barra invertida, despues la comilla."""
    return text.replace("\\", "\\\\").replace('"', '\\"')


def jpath(path):
    """Una ruta como aparece dentro de un texto JSON (sin las comillas)."""
    return json.dumps(path)[1:-1]


def build_script(json_text=None, line=None, extra=None, replace=None):
    lines = list(lines0)
    lines[IDX] = line if line is not None else PREFIX + py_escape(json_text) + '"'
    text = "\n".join(lines)
    if replace:
        assert text.count(replace[0]) == 1, replace[0]
        text = text.replace(replace[0], replace[1])
    if extra:
        text = text.replace(lines[IDX] + "\n", lines[IDX] + "\n" + extra + "\n", 1)
    return text


def relocate_json(new_folder, folder_in_payload=None):
    target = folder_in_payload if folder_in_payload is not None else new_folder
    assert json0.count(jpath(folder0)) == 1
    return json0.replace(jpath(folder0), jpath(target))


def relocate_manifest(new_folder):
    assert manifest0.count(jpath(folder0)) == 1
    return manifest0.replace(jpath(folder0), jpath(new_folder))


def write_pkg(pkg_dir, script_text, manifest_text):
    os.makedirs(pkg_dir)
    open(os.path.join(pkg_dir, "rackcad-ram-build.py"), "wb").write(script_text.encode("utf-8"))
    open(os.path.join(pkg_dir, "manifest.json"), "wb").write(manifest_text.encode("utf-8"))
    open(os.path.join(pkg_dir, "structural-loads.json"), "wb").write(real_bytes["structural-loads.json"])


def case_dir(name):
    return os.path.join(root, name, "pkg-01")


cases = []


def add(name, kind, description, script_text, manifest_text, pkg_arg, typed, physical=None):
    write_pkg(physical or pkg_arg, script_text, manifest_text)
    cases.append({"case": name, "kind": kind, "description": description, "pkg": pkg_arg, "typed": typed})


# Positivos
p = case_dir("P01-reubicado")

p = os.path.join(root, "P02 carpeta \u00f1 espacio", "pkg-01")

p = case_dir("N04-escape-menos-dos")

add("N04-escape-menos-dos", "negativo", "sin ningun escape de la barra invertida (solo comillas)",
    build_script(line=PREFIX + raw_json.replace('"', '\\"') + '"'), relocate_manifest(p), p, p)
p = case_dir("N05-escape-mas-uno")

    build_script(relocate_json(p).replace(jpath(p), jpath(jpath(p)))), relocate_manifest(p), p, p)
p = case_dir("N06-mayusculas")

results, ok_all = [], True
for c in cases:
    rep = os.path.join(root, "reports", c["case"] + "-v2.json")
    proc = subprocess.run([sys.executable, "-I", v2, c["pkg"], expected, c["typed"], rep], capture_output=True, text=True, encoding="utf-8")
    r = json.load(open(rep, encoding="utf-8"))
    v11 = [x for x in r["checks"] if x["id"] == "V-11"][0]
    want = "PASS" if c["kind"] == "positivo" else "FAIL"
    good = v11["result"] == want
    if c["case"].startswith("N11"):

        c["sentinelAbsent"] = not os.path.exists(sentinel)
    ok_all &= good
    results.append(dict(c, expectedV11=want, observedV11=v11["result"], ok=good, v11Detail=v11["detail"], validatorExit=proc.returncode))
    print("%-28s %-9s V-11 esperado=%s observado=%s %s" % (c["case"], c["kind"], want, v11["result"], "OK" if good else "**NO**"))
