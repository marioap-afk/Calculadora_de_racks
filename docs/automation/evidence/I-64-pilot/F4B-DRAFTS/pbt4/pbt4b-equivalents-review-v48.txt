    [Theory]
    [InlineData("a-veto")]

    [InlineData("an-admitted-intention")]

    [InlineData("a-drain")]
    [InlineData("a-drain-with-the-stale-hint")]
    [InlineData("a-note-of-an-edit")]
    public void EachOperationThatCanChangeADraftEndsTheMarkSoTheSessionIsRetiredConservatively(string operation)
    {
        Action<Env>? before = operation switch
        {
            "a-check-that-finds-a-conflict" or "a-check-that-recovers" => env => env.UnknownPilot(),
            "a-reload" => env => env.StalePilot(),
            _ => null,
        };

        Action<Env> between = operation switch
        {
            "a-veto" => env =>
            {
                var dirty = env.Seed("R1");
                dirty.SetCurrentDraftState(dirty.CurrentDraftState with { Name = "Edited" });
                Assert.Equal(CloseDecision.Veto, env.Advisor.AdviseClose(env.Session));
                env.Core.OpenDrafts.DiscardByUser(env.Session, "R1");
            },
            "a-failure-of-the-evaluation" => env =>
            {
                var comparer = new DraftThrowingComparer();
                env.Seed("R1", comparer);
                comparer.Throw = true;
                Assert.Equal(CloseDecision.Veto, env.Advisor.AdviseClose(env.Session));
                env.Core.OpenDrafts.DiscardByUser(env.Session, "R1");
            },
            "an-admitted-intention" => env => Assert.Equal(EditIntentStatus.Admitted, env.Core.TryBeginPilotEdit(env.Request()).Status),
            "a-check-that-finds-a-conflict" => env =>
            {
                env.Reader.Answer = FullEditIntentKit.MemberBecameBlocking;
                Assert.Equal(EditCheckStatus.Conflict, env.Core.TryCheckPilotEdit(env.Request()).Status);
            },
            "a-check-that-recovers" => env => Assert.Equal(EditCheckStatus.Recovered, env.Core.TryCheckPilotEdit(env.Request()).Status),
            "a-reload" => env => Assert.Equal(EditIntentStatus.Reloaded, env.Core.TryReloadPilotEdit(env.Request()).Status),
            "a-discard-of-the-host" => env => Assert.Equal(EditDiscardStatus.Discarded, env.Core.DiscardPilotEdits(env.Session).Status),
            "a-drain" => env => env.Core.OnDrained(Outcome(env.Session)),
            "a-drain-with-the-stale-hint" => env => env.Core.OnDrained(Outcome(env.Session, DrainOperation.MarkDraftsPossiblyStale)),
            _ => env => env.Core.NotePilotEdit(env.Session),
        };

        Assert.Equal("conservative", PathAfter(before, between));
    }

    // ------------------------------------------------------------------ INV-F4T3A-01 to 03: the host core

    private static IEnumerable<string> HostViolations(string host)
    {
        // The interface of the advisor stays the first base (WorkspaceLifecycleGuardTests), and the port is the second.
        if (!Regex.IsMatch(host, @"public\s+sealed\s+class\s+WorkspaceHostCore\s*:\s*IWorkspaceLifecycleAdvisor\s*,\s*IWorkspacePilotEditPort\b"))
            yield return "the host is not : IWorkspaceLifecycleAdvisor, IWorkspacePilotEditPort";

        // Explicit implementations only: the public surface of the host does not grow.
        foreach (var (name, _) in Operations)
        {
            if (Src.Count(host, HostSignature(name)) != 1) yield return "the host does not implement " + name + " explicitly exactly once";
            if (Regex.IsMatch(host, @"\bpublic\s+[\w<>,\s?]+\s+" + name + @"\s*\(")) yield return "the host exposes " + name + " publicly";
        }

        // The adapter of the pilot is static, never a field of instance (H-10).
        if (Regex.IsMatch(host, @"private\s+(?:readonly\s+)?PushBackFullFieldAdapter\s+_\w+")) yield return "the adapter of the field is a field of instance";
        if (Src.Count(host, @"private\s+static\s+readonly\s+PushBackFullFieldAdapter\s+\w+") != 1) yield return "the adapter of the field is not the one static field";

        // Only the three reads read the drawing (C03): the other nine operations never name the reader, the probe or the gate of reading.
        // Row 5: the request of the update is the one exception, with its own rules below.
        foreach (var (name, _) in Operations.Where(o => !ReadOperations.Contains(o.Name) && !UpdateOperations.Contains(o.Name)))
        {
            var body = TryBody(host, HostSignature(name));
            if (body == null) continue;
            if (Regex.IsMatch(body, @"_editIntentReader|ReadIntent|_reader\b|PassGate|_probe\b|_bridge\b")) yield return name + " reaches a reader, the probe or the bridge";
        }

        // The request of the update: it passes the gate of the document, the only place of the rest, and it is the one caller of the executor.
        foreach (var name in UpdateOperations)
        {
            var body = TryBody(host, HostSignature(name));
            if (body == null) continue;
            if (!Regex.IsMatch(body, @"\bPassGate\s*\(")) yield return name + " does not pass the gate of the document";
            if (Regex.IsMatch(body, @"_editIntentReader|ReadIntent|_reader\b")) yield return name + " reads the drawing";
            if (Src.Count(body, @"\.TryRequest\s*\(") != 1) yield return name + " does not ask the executor exactly once";
            if (Regex.IsMatch(body, @"TryUpdatePilot|\.Execute\s*\(")) yield return name + " runs the update itself";
        }

        // The lazy bound of the ticket: it only reads the count of foreign commands of the bridge, and it never touches the own mutation,
        // the reader, the probe or the gate (RR N-04). The same for the progress, which is memory only.
        var expire = TryBody(host, @"private\s+UpdateTicket\?\s+ExpireStaleUpdate\s*\(");
        if (expire == null) yield return "ExpireStaleUpdate is missing";
        else
        {
            if (Regex.IsMatch(expire, @"CloseOwnMutation|OpenOwnMutation|_editIntentReader|_reader\b|_probe\b|PassGate")) yield return "ExpireStaleUpdate reaches more than the count of foreign commands";
            if (Regex.Matches(expire, @"_bridge\.(\w+)").Cast<Match>().Any(m => m.Groups[1].Value != "ForeignCommandStartsOf")) yield return "ExpireStaleUpdate names a member of the bridge other than ForeignCommandStartsOf";
            if (!Regex.IsMatch(expire, @"_bridge\.ForeignCommandStartsOf\s*\(")) yield return "ExpireStaleUpdate does not read the count of foreign commands";
        }

        var progress = TryBody(host, HostSignature("TakeUpdateProgress"));
        if (progress != null && Regex.IsMatch(progress, @"OwnMutation|_bridge\b|_probe\b|PassGate|_lastAllowedClose")) yield return "TakeUpdateProgress mutates the bridge or looks at the document";

        // While an update is in flight (D-T4C-08, D-T4C-10; M-01): the check of the flight comes after the authorization (Authorize or
        // PassGate) and after the mark of the reads where the body has it. A body without any authorization has no order to keep: it trips.
        foreach (var (name, signature, markInBody) in FlightBodies)
        {
            var body = TryBody(host, signature);
            if (body == null) { yield return name + " is missing"; continue; }

            var check = FlightCheck.Match(body);
            var mark = body.IndexOf("_pilotReadInProgress", StringComparison.Ordinal);
            var authorize = FirstAuthorization(body);
            if (!check.Success) yield return name + " does not look at the update in flight";
            else if (authorize < 0 || check.Index < authorize) yield return name + " looks at the update in flight before its authorization";
            else if (markInBody && (mark < 0 || check.Index < mark)) yield return name + " looks at the update in flight before the mark of the reads";
        }

        // Memory only: no authorization at all.
        foreach (var name in MemoryOperations)
        {
            var body = TryBody(host, HostSignature(name));
            if (body != null && Regex.IsMatch(body, @"\bAuthorize\b|_pilotReadInProgress")) yield return name + " authorizes or looks at the mark of the reads";
        }

        // The operations that change something: authorization, then the mark of the reads, then the effect.
        foreach (var name in MutatingOperations)
        {
            var body = TryBody(host, HostSignature(name));
            if (body == null) continue;

            // The request of the update authorizes through the gate of the document, which ends with the authorization of the session.
            var authorizes = Regex.IsMatch(body, @"\bAuthorize\s*\(") || (UpdateOperations.Contains(name) && Regex.IsMatch(body, @"\bPassGate\s*\("));
            if (!authorizes) yield return name + " does not authorize";
            if (!Regex.IsMatch(body, @"if\s*\(\s*_pilotReadInProgress\s*\)")) yield return name + " does not look at the mark of the reads";
        }

        // NotePilotEdit (or the discard of the session) in every operation that changes a draft (D-T3-02).
        foreach (var name in new[] { "SetPendingText", "CommitField", "CrossBoundary", "DiscardRack" })
        {
            var body = TryBody(host, HostSignature(name));
            if (body != null && !Regex.IsMatch(body, @"\bNotePilotEdit\s*\(")) yield return name + " does not call NotePilotEdit";
        }

        var document = TryBody(host, HostSignature("DiscardDocument"));
        if (document != null && !Regex.IsMatch(document, @"\bDiscardPilotEdits\s*\(")) yield return "DiscardDocument does not delegate in DiscardPilotEdits";

        // A non-blocking boundary needs no authorization: it answers Kept before the one authorization of the blocking ones.
        var cross = TryBody(host, HostSignature("CrossBoundary"));
        if (cross != null)
        {
            var kept = cross.IndexOf("PilotBoundaryDecision.Kept", StringComparison.Ordinal);
            var authorize = cross.IndexOf("Authorize(", StringComparison.Ordinal);
            if (kept < 0 || authorize < 0 || kept > authorize) yield return "CrossBoundary does not keep the non-blocking boundaries before it authorizes";
        }

        // The ways the host contains an exception: a value with a reason, never the degradation of the session.
        foreach (var name in new[] { "SetPendingText", "CommitField", "CrossBoundary", "DiscardRack", "DiscardDocument", "GetField", "GetRacksAtRisk", "RequestUpdate", "TakeUpdateProgress" })
        {
            var body = TryBody(host, HostSignature(name));
            if (body == null) continue;
            if (!Regex.IsMatch(body, @"catch\s*\(\s*Exception")) yield return name + " does not contain its exception";
            if (Regex.IsMatch(body, @"_faulted|\bContain\s*\(|\bthrow\s*;")) yield return name + " degrades the session or rethrows";
        }
    }

    [Fact]
    public void AnExceptionOfTheReaderIsFailedWithTheMessageReportedAndTheMarkAndTheSessionAreNotTouched()

        {
            var rig = new Rig();

            draft.MarkPossiblyStale();
            rig.Reader.Throw = new InvalidOperationException("the reader threw");

            string? message;
            string context;
            switch (operation)
            {
                case "confirm":
                    var confirm = rig.Port.TryConfirmBase(rig.Request());
                    Assert.Equal(EditConfirmStatus.Failed, confirm.Status);
                    message = confirm.Reason;
                    context = "WorkspaceHostCore.TryConfirmPilotBase";

                    Assert.Equal(EditIntentStatus.Failed, keep.Status);
                    message = keep.Reason;
                    context = "WorkspaceHostCore.TryKeepPilotEdit";
                    break;
                default:
                    var discard = rig.Port.TryDiscardAndReload(rig.Request());
                    Assert.Equal(EditIntentStatus.Failed, discard.Status);
                    message = discard.Reason;
                    context = "WorkspaceHostCore.TryDiscardAndReloadPilotEdit";
                    break;
            }

            Assert.Equal("the reader threw", message);
            var fault = Assert.Single(rig.Host.Faults);
            Assert.Equal(context, fault.Context);
            Assert.Equal(WorkspaceHostStatus.Ready, rig.Core.GetSnapshot().Status);
            Assert.False(rig.Host.Bridge.Core.IsDegraded(rig.Session));
            Assert.Equal(DraftStatus.Dirty, draft.Status);
            Assert.True(draft.IsPossiblyStale);
            // The mark of the read was released by the finally: a second call reads again.
            rig.Reader.Throw = null;
            Assert.NotEqual("Hay una lectura del piloto en curso.", rig.Port.TryBeginEdit(rig.Request()).Reason);
        }
    }

    {
        var rig = new Rig();

        draft.MarkPossiblyStale();
        var nested = new List<(string Operation, string Status, string? Reason)>();
        rig.Reader.Answer = rack =>

            var keep = rig.Port.TryKeepMine(rig.Request());
            var discard = rig.Port.TryDiscardAndReload(rig.Request());
            var begin = rig.Port.TryBeginEdit(rig.Request());
            nested.Add(("confirm", confirm.Status.ToString(), confirm.Reason));
            nested.Add(("keep", keep.Status.ToString(), keep.Reason));
            nested.Add(("discard", discard.Status.ToString(), discard.Reason));
            nested.Add(("begin", begin.Status.ToString(), begin.Reason));
            return FullEditIntentKit.Admitted(rack);
        };

        var results = new string[3];
        results[0] = rig.Port.TryConfirmBase(rig.Request()).Status.ToString();

        results[1] = rig.Port.TryKeepMine(rig.Request()).Status.ToString();

        {

        var cases = new Dictionary<string, string>
        {
            // The frontal view of side B in a rack that has only side A.
            ["side-b"] = FullPrepareKit.Frontal(section: 2),

            ["corrupt-descriptor"] = PilotKit.Text("pb-01-base"),
            ["section-out-of-range"] = FullPrepareKit.Frontal(section: 9),
        };

        foreach (var (name, text) in cases)
        {
            var members = FullPrepareKit.BaseOf(("A1", originFrontal), ("B2", text));

            FullPrepareKit.SetScale(draft, 2.25);

            // M-5 (I-64 F4-T4B): a descriptor that does not decode has no address to ask about, so it has its own text and is never reported as a
            // view that is "no longer available".
            var corrupt = name is "corrupt-descriptor" or "section-out-of-range";
            // I-64 F4B PB-T4b (D-S1; D-PBT3-09 (e)): with the full model, the frontal of side B of a rack without side B has its own text.
            Assert.StartsWith((corrupt ? PushBackPilotUpdatePlanner.ViewAddressPrefix : PushBackPilotUpdatePlanner.FrontalBOfOneWayPrefix) + "B2", result.Reason, StringComparison.Ordinal);

        }

        // I-64 F4B PB-T4b (D-S1; D-PB-04, D-PBT3-09 (e) and D-PBT4-07): the full model erases the obsolete lateral, so the phantom is not
        // a view that is no longer available: the update is Ready, the phantom is erased and it is never in the plan of the redrawn views.
        var phantomMembers = FullPrepareKit.BaseOf(("A1", originFrontal), ("B2", FullPrepareKit.Lateral(FullPrepareKit.MissingPost)));
        var phantomDraft = FullPrepareKit.DraftOf(phantomMembers, originFrontal);
        FullPrepareKit.SetScale(phantomDraft, 2.25);
        var phantom = FullPrepareKit.Prepare(phantomDraft, FullPrepareKit.Reread(phantomMembers));
        Assert.Equal(PushBackPilotPrepareKind.Ready, phantom.Kind);
        Assert.Equal(new[] { "A1" }, phantom.Plan!.Members.Select(m => m.Key).ToArray());
        Assert.Equal(new[] { "B2" }, phantom.Plan!.Erasures.ToArray());

        Assert.NotEqual(origin, originFrontal);
    }

        var design = new PushBackCompositeEditorAssembler(PilotKit.Catalog).BuildDesign(composition.Editor!, composition.Inputs!);
        var store = new RackProjectStore();

        foreach (var member in plan.Members)
        {
            var source = new RackEmbedStore().Deserialize(members.Entries[member.Key].EnvelopeText!)!;
            var expectedDesign = store.Serialize(RackProject.ForPushBack(design).WithSourceMetadataFrom(store.Deserialize(source.Design)));
            var envelope = new RackEmbedStore().Deserialize(member.EnvelopeText)!;

            Assert.Equal(expectedDesign, member.DesignText);
            Assert.Equal(member.DesignText, envelope.Design);
            Assert.Equal(RackEmbedDocument.KindPushBack, envelope.Kind);
            Assert.Equal(source.Id, envelope.Id);
            // Name never comes from the draft: it is the name of the envelope that was re-read for that view.
            Assert.Equal(source.Name, envelope.Name);
            Assert.Equal("Name of the rack", envelope.Name);
            Assert.Equal(member.View, envelope.View);
            Assert.Equal(member.Section, envelope.Section);
            Assert.NotNull(envelope.ExtensionData);
            Assert.Contains(envelope.ExtensionData!.Keys, k => k.StartsWith("x", StringComparison.Ordinal));
            // The scale is the only thing the pilot changed in the design that goes out.
            Assert.Contains("2.25", member.DesignText, StringComparison.Ordinal);
        }

        Assert.Equal(plan.Members.Select(m => m.EnvelopeText).ToArray(), plan.PostExpectation.Envelopes.Select(e => e.Text).ToArray());
        Assert.Equal(plan.Members.Select(m => m.Key).ToArray(), plan.PostExpectation.Envelopes.Select(e => e.Key).ToArray());

tests/RackCad.Tests/Workspace/WorkspacePilotUpdateIntegrationGuardTests.cs :: ayudante G4Violations
====================================================================================================

    private static IEnumerable<string> G4Violations(IReadOnlyList<(string Path, string Code)> product)
    {
        // M-07: the IDENTIFIER, as a whole word and outside the text of a string, so that a group of methods (a delegate that is the method itself)
        // is seen too: it appears in the core twice (its declaration and its one call) and nowhere else.
        var callers = product.Where(f => Regex.IsMatch(WithoutStrings(f.Code), @"\bTryUpdatePilot\b")).Select(f => f.Path).ToArray();
        if (!callers.SequenceEqual(new[] { CorePath })) yield return "the callers are in " + string.Join(", ", callers);

        var core = product.Single(f => f.Path == CorePath).Code;
        var run = Src.BodyOf(core, @"private\s+sealed\s+class\s+PilotUpdateRun\b");
        var execute = Src.BodyOf(run, @"public\s+void\s+Execute\s*\(");
        if (Src.Count(core, @"\.TryUpdatePilot\s*\(") != 1 || Src.Count(execute, @"\.TryUpdatePilot\s*\(") != 1) yield return "TryUpdatePilot is not called exactly once, from Execute of the run of the core";
        if (Src.Count(WithoutStrings(core), @"\bTryUpdatePilot\b") != 2) yield return "the identifier TryUpdatePilot appears other than as its declaration and its one call";

        var runCallers = product.Where(f => Regex.IsMatch(f.Code, @"PushBackPilotUpdateRun\.ExecuteFull\s*\(")).Select(f => f.Path).ToArray();
        if (!runCallers.SequenceEqual(new[] { CorePath })) yield return "PushBackPilotUpdateRun.ExecuteFull is named in " + string.Join(", ", runCallers);
    }

    // ------------------------------------------------------------------ the answer of a close that was not accepted is never a success (M-01)

    [Fact]
    public void AnUpdateOfTheCoreThatRunsOverASlotThatWasTakenAwayNeverAnswersAnApplied()
    {
        // The slot is taken from the draft while the run is in its write: CompleteApply finds no update to close (NotApplying).
        var rig = new Rig();
        var draft = rig.Open();
        rig.Writer.HookAt["Seal"] = () => Assert.True(draft.CompleteApply(DraftApplyOutcome<PushBackFullDraftState>.NotAppliedFailure("taken away")).Accepted);

        var outcome = rig.Core.TryUpdatePilot(rig.Request(), rig.Writer);

        Assert.NotEqual(EditUpdateStatus.Applied, outcome.Status);
        Assert.NotEqual(EditUpdateStatus.AppliedWithWarnings, outcome.Status);
        Assert.Equal(EditUpdateStatus.ClosedInError, outcome.Status);
        Assert.Equal(EditUpdateStatus.Applied, outcome.Classification);
        Assert.Equal(PushBackPilotHostTexts.UpdateNotClosedReason, outcome.Reason);
    }

====================================================================================================

    [Fact]
    public void DuringTheFlightTheFiveOperationsThatCouldChangeTheDraftAnswerBusyOrWaitingAndTheDraftDoesNotChange()
    {
        var rig = new Rig();
        var draft = rig.Open();
        rig.Kit.Primitives.Mode = PilotHostMode.Deferred;
        rig.Start();
        var before = Snapshot(draft);
        var reads = rig.Reader.Calls.Count;

        var set = rig.Port.SetPendingText(rig.Origin(), RackA, "4");
        var commit = rig.Port.CommitField(rig.Origin(), RackA);
        var keep = rig.Port.TryKeepMine(rig.Request());
        var discard = rig.Port.TryDiscardAndReload(rig.Request());
        var confirm = rig.Port.TryConfirmBase(rig.Request());

        Assert.Equal(PilotFieldStatus.Busy, set.Status);
        Assert.Equal(PushBackPilotHostTexts.UpdateInFlightReason, set.Reason);
        Assert.Equal(PilotFieldStatus.Busy, commit.Status);
        Assert.Equal(EditIntentStatus.Waiting, keep.Status);
        Assert.Equal(PushBackPilotHostTexts.UpdateInFlightReason, keep.Reason);
        Assert.Equal(EditIntentStatus.Waiting, discard.Status);
        Assert.Equal(EditConfirmStatus.Waiting, confirm.Status);
        Assert.Equal(PushBackPilotHostTexts.UpdateInFlightReason, confirm.Reason);
        Assert.Equal(before, Snapshot(draft));
        Assert.Equal(reads, rig.Reader.Calls.Count);

        // After the end they are admitted again.
        rig.Kit.Primitives.RunDeferred();
        rig.Take();
        Assert.NotEqual(PilotFieldStatus.Busy, rig.Port.SetPendingText(rig.Origin(), RackA, "4").Status);
    }

tests/RackCad.Tests/Workspace/WorkspacePilotUpdateRequestTests.cs :: TheOutcomeThatT4BGivesIsKeptWithTheSameStatusReasonDifferencesAndWarningsAsThePlainCallOfTheUpdate
====================================================================================================

    [Theory]
    [MemberData(nameof(TheOutcomesOfT4B))]
    public void TheOutcomeThatT4BGivesIsKeptWithTheSameStatusReasonDifferencesAndWarningsAsThePlainCallOfTheUpdate(string name)
    {
        var direct = new Rig();
        direct.Open();
        Arrange(direct.Writer, name);
        var expected = direct.Core.TryUpdatePilot(direct.Request(), direct.Writer);

        var rig = new Rig();
        rig.Open();
        Arrange(rig.Writer, name);
        rig.Start();
        var kept = rig.Take().Outcome;

        Assert.Equal(expected.Status, kept.Status);
        Assert.Equal(expected.Reason, kept.Reason);
        Assert.Equal(expected.Differences.ToArray(), kept.Differences.ToArray());
        Assert.Equal(expected.Warnings.ToArray(), kept.Warnings.ToArray());
    }

    [Fact]
    public void TheTextsOfTheGateOfTheUpdateAreTheConstantsOfTheRunWithTheirOldWording()
    {
        var modal = new Rig();
        modal.Open();
        modal.Core.SetModalRackCadActive(true);
        var rejected = modal.Core.TryUpdatePilot(modal.Request(), modal.Writer);
        Assert.Equal(EditUpdateStatus.Rejected, rejected.Status);
        Assert.Equal(PushBackPilotUpdateRun.ModalOpenReason, rejected.Reason);

        var none = new Rig();
        none.Open();
        none.Host.Probe.ActiveDocumentNativeId = null;
        var waiting = none.Core.TryUpdatePilot(none.Request(), none.Writer);
        Assert.Equal(PushBackPilotUpdateRun.NoActiveDocumentReason, waiting.Reason);
    }
