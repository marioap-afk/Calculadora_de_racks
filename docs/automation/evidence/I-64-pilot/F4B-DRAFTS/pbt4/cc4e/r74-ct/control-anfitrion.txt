                // Without work for the rack the edit intention begins it, with its one read, its admission and its diagnostic. Over a clean
                // work marked as possibly stale the lazy reload of "Editar" runs first (I-64 F4B PB-T4e, D-PBT4-22): the same method, with its
                // one read, before the editor opens; a reload that does not reload the work does not open the editor and carries its outcome.
                // Either way the gesture reads the drawing at most once, and never while the editor is open.
                var draft = FindPilotDraft(request.Session, request.RackId);
                if (draft == null)
                {
                    var begun = TryBeginPilotEdit(request);
                    if (begun.Status != EditIntentStatus.Admitted) return EditAllOutcome.NotBegun(begun);
                    draft = FindPilotDraft(request.Session, request.RackId);
                    if (draft == null) return EditAllOutcome.Failed(EditAllBusyDraftReason);
                }
                else if (draft.NeedsLazyReload)
                {
                    var reload = TryReloadPilotEdit(request);
                    if (reload.Status != EditIntentStatus.Reloaded) return EditAllOutcome.NotBegun(reload);
                }
