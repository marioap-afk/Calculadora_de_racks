function Sha([string]$p) { (Get-FileHash -LiteralPath $p -Algorithm SHA256).Hash.ToLower() }
function G { $r = & git -C $Wt @args; if ($LASTEXITCODE -ne 0) { throw "git $args" }; ($r -join "`n").Trim() }
function Chk([string]$name, [bool]$ok, [string]$detail) { $L.Add(("{0} {1} {2}" -f ($(if ($ok) { 'OK  ' } else { 'FAIL' })), $name, $detail)); if (-not $ok) { $fail.Add($name) } }
function Manifest([string]$dir, [string]$out) {
    $rows = Get-ChildItem -LiteralPath $dir -Recurse -File | Sort-Object { $_.FullName.Substring($dir.Length + 1).Replace('\', '/') } -Culture ([Globalization.CultureInfo]::InvariantCulture) | ForEach-Object {
        "{0}  {1}  {2}" -f $_.FullName.Substring($dir.Length + 1).Replace('\', '/'), (Sha $_.FullName), $_.Length
    }
    [IO.File]::WriteAllText($out, (($rows -join "`n") + "`n"), $utf8)
    @($rows).Count
}
function BlobSha([string]$rev, [string]$path) {
    $psi = New-Object Diagnostics.ProcessStartInfo 'git', "-C `"$Wt`" cat-file blob ${rev}:$path"
    $psi.RedirectStandardOutput = $true; $psi.UseShellExecute = $false
    $pr = [Diagnostics.Process]::Start($psi); $ms = New-Object IO.MemoryStream; $pr.StandardOutput.BaseStream.CopyTo($ms); $pr.WaitForExit()
    $h = [Security.Cryptography.SHA256]::Create().ComputeHash($ms.ToArray())
    -join ($h | ForEach-Object { $_.ToString('x2') })
}
