# W-R1a (I-65): copia las capturas de la sesion a la corrida; recorta lo identificable que no es necesario (barra de titulo con la cuenta de Autodesk, Explorador del usuario, barra de tareas).
param([Parameter(Mandatory)][string]$ToolResults, [Parameter(Mandatory)][string]$Out)
$ErrorActionPreference = 'Stop'
Add-Type -AssemblyName System.Drawing
New-Item -ItemType Directory -Force -Path $Out | Out-Null
function Src([string]$suffix) { $f = @(Get-ChildItem -LiteralPath $ToolResults -Filter "mcp-computer-use-blob-*-$suffix.jpg"); if ($f.Count -ne 1) { throw "captura ${suffix}: $($f.Count)" }; $f[0].FullName }
function Copy1([string]$suffix, [string]$name) { Copy-Item -LiteralPath (Src $suffix) -Destination (Join-Path $Out $name) -Force }
function Crop([string]$suffix, [string]$name, [int]$x, [int]$y, [int]$w, [int]$h) {
  $img = [System.Drawing.Image]::FromFile((Src $suffix))
  if ($x + $w -gt $img.Width -or $y + $h -gt $img.Height) { $iw = $img.Width; $ih = $img.Height; $img.Dispose(); throw "recorte fuera de ${suffix}: ${iw}x${ih}" }
  $bmp = New-Object System.Drawing.Bitmap $w, $h
  $g = [System.Drawing.Graphics]::FromImage($bmp); $g.DrawImage($img, (New-Object System.Drawing.Rectangle 0, 0, $w, $h), (New-Object System.Drawing.Rectangle $x, $y, $w, $h), [System.Drawing.GraphicsUnit]::Pixel)
  $g.Dispose(); $img.Dispose(); $bmp.Save((Join-Path $Out $name), [System.Drawing.Imaging.ImageFormat]::Jpeg); $bmp.Dispose() }
# AutoCAD a escala 0.5 (728x409): sin barra de titulo (cuenta), sin paleta de propiedades ni barra de tareas
function Acad05([string]$s, [string]$n) { Crop $s $n 0 14 565 358 }
# AutoCAD a escala 0.6 (874x491)

function Acad06P([string]$s, [string]$n) { Crop $s $n 0 16 874 430 }
# Editor WPF a escala 1 (1456x819): solo la ventana del editor (el Explorador del usuario queda fuera)
function Editor1([string]$s, [string]$n) { Crop $s $n 0 0 1157 610 }
