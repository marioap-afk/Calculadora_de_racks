# I-65 W-R2: preflight de SOLO LECTURA (Owner §4, puntos 1, 5, 6, 8 y 10; procedimientos §4.1 WR2-PR-03..06). No abre RAM.
# Unicas escrituras: archivos nuevos en <Run>\wr2-preflight\. Sale con 3 si alguna comprobacion falla.
# Uso: pwsh -NoProfile -File wr2_preflight.ps1 -Run D:\r65-smoke\4a03b65b\wr1-R20261029T1633Z -Worktree <worktree> -Tools <scratchpad>\i65
param([Parameter(Mandatory)][string]$Run, [Parameter(Mandatory)][string]$Worktree, [Parameter(Mandatory)][string]$Tools)
$ErrorActionPreference = 'Stop'
$out = Join-Path $Run 'wr2-preflight'
if (Test-Path -LiteralPath $out) { throw "ya existe $out" }
New-Item -ItemType Directory -Path $out | Out-Null
$L = New-Object Collections.Generic.List[string]
$bad = New-Object Collections.Generic.List[string]
function Sha([string]$p) { (Get-FileHash -Algorithm SHA256 -LiteralPath $p).Hash.ToLower() }
function TSha([string]$s) { [Convert]::ToHexString([Security.Cryptography.SHA256]::HashData([Text.Encoding]::UTF8.GetBytes($s))).ToLower() }
function Check([string]$id, [bool]$ok, [string]$text) { $L.Add(("{0} {1} {2}" -f $(if ($ok) { 'OK  ' } else { 'FAIL' }), $id, $text)); if (-not $ok) { $bad.Add($id) } }
$L.Add("time_utc=" + [DateTime]::UtcNow.ToString('o'))

# 1. AutoCAD y RAM cerrados
$n = @(Get-Process | Where-Object { $_.ProcessName -match '^(acad|accoreconsole|RAMElements)$' }).Count
Check 'P1.procesos' ($n -eq 0) "acad/accoreconsole/RAMElements=$n"

# Herramientas del Coordinator (procedimientos §1.3)
$tools = [ordered]@{ 'w2_readiness.ps1' = '466f7a933042b34e423eb021cf99dd942c57a45d62391db306935d0297bab827'; 'w2_hostsnap.ps1' = '6904a43965de69b33b78c304cd57d1958b31b2018121906cf22d034724ff5485'; 'session_custody.ps1' = '5487544aec2685c8554145b46d68bd4fe8fe9545e0b1628deee84d7fc080491b'; 'readback_dump_contrast.py' = '0790930297d221ddefeadaef8e04611858b2e81446717155e754d34155114bb1'; 'secrets_scan.py' = 'bcccaf9123923eca9f998c078ba3850d6d934c133d3d170ce1b969abe4fa9fb3'; 'w1_idle.ps1' = 'ac61c01bae9975e335b5632bb0ed6066f341f1005cb44af3fd30c06e690fa760' }
foreach ($t in $tools.Keys) { $s = Sha (Join-Path $Tools $t); Check "T.$t" ($s -eq $tools[$t]) $s }
# Herramienta del producto (comparador): manifiesto recalculado con la funcion Manifest de fc_identity.ps1
$tool = 'D:\r65-smoke\4a03b65b\ramtool'
$rows = Get-ChildItem -LiteralPath $tool -Recurse -File | Sort-Object { $_.FullName.Substring($tool.Length + 1).Replace('\', '/') } -Culture ([Globalization.CultureInfo]::InvariantCulture) | ForEach-Object { "{0}  {1}  {2}" -f $_.FullName.Substring($tool.Length + 1).Replace('\', '/'), (Sha $_.FullName), $_.Length }
$ms = TSha (($rows -join "`n") + "`n")
Check 'T.ramtool' ($ms -eq '4fbe3ea5be1790c20c2673339b9ca2fba26bdac8f59472686952474fb33eb343' -and @($rows).Count -eq 11) "files=$(@($rows).Count) manifest_sha256=$ms"

# 5. Paquetes, inyecciones y scripts = lista custodiada de W-R1a (evidencia §120)
$list = Join-Path $Worktree 'docs\automation\evidence\I-65-ready\wr1a-R20261029T1633Z\package-dwg-and-originals-sha256.txt'
$dirs = 'pkg-OV01', 'pkg-OV02', 'pkg-OV04', 'inj-OV05', 'inj-OV06', 'inj-OV07a', 'inj-OV07b', 'dump-OV01-gen', 'dump-OV02-gen'
$exp = @{}
foreach ($line in (Get-Content -LiteralPath $list)) { if ($line -match '^([0-9a-f]{64})  (\S+)/(\S+)$' -and $dirs -contains $Matches[2]) { $exp[$Matches[2] + '/' + $Matches[3]] = $Matches[1] } }
Check 'P5.lista' ($exp.Count -eq 23) "entradas=$($exp.Count) (7 paquetes x 3 + 2 scripts de volcado)"
foreach ($d in $dirs) {
  $files = @(Get-ChildItem -File -Force -LiteralPath (Join-Path $Run $d) | Sort-Object Name)
  $want = @($exp.Keys | Where-Object { $_.StartsWith("$d/") } | ForEach-Object { $_.Substring($d.Length + 1) } | Sort-Object)
  $have = @($files | ForEach-Object { $_.Name })
  $same = ($want -join ',') -ceq ($have -join ',')
  foreach ($f in $files) { if ($exp.ContainsKey("$d/" + $f.Name)) { $s = Sha $f.FullName; if ($s -ne $exp["$d/" + $f.Name]) { $same = $false; $L.Add("     $d/$($f.Name) $s distinto") } } }
  Check "P5.$d" $same ("archivos=" + ($have -join ','))
}
# 6. Destinos sin lecturas de vuelta ni volcados
foreach ($d in $dirs) {
  $f = $(if ($d -like 'dump-*') { 'ram-dump.json' } else { 'ram-readback.json' })
  Check "P6.$d" (-not (Test-Path -LiteralPath (Join-Path $Run "$d\$f"))) "$f existe=$(Test-Path -LiteralPath (Join-Path $Run "$d\$f"))"
}
# 8. R-D2: volcados manuales, plantilla de volcado y especificaciones
$m1 = 'D:\r65-smoke\51fc82f1\run-R20261016T1918Z\p2-R20261016T2147Z\dump-man'
$m2 = 'D:\r65-smoke\bd1fcf8a\f5-R20261019T1750Z\dump-OV02-man'
Check 'P8.manual-OV01' ((Sha "$m1\ram-dump.json") -eq 'c4a98fb0ea798031de2b3052d81eae89f42b731a92f06dbfa97f148ade9ba23b') (Sha "$m1\ram-dump.json")
Check 'P8.manual-OV02' ((Sha "$m2\ram-dump.json") -eq '12c89479fa2bf54adde27a1d0c4affc18bbd0469e77cd72c1fb5d0335910aef2') (Sha "$m2\ram-dump.json")
$tpl = @(Get-Content -LiteralPath "$m1\rackcad-ram-dump.py")
$tplFolder = 'D:\\\\r65-smoke\\\\51fc82f1\\\\run-R20261016T1918Z\\\\p2-R20261016T2147Z\\\\dump-man'
foreach ($d in 'dump-OV01-gen', 'dump-OV02-gen') {
  $cur = @(Get-Content -LiteralPath (Join-Path $Run "$d\rackcad-ram-dump.py"))
  $diff = @(); for ($i = 0; $i -lt [Math]::Max($cur.Count, $tpl.Count); $i++) { if ($cur[$i] -cne $tpl[$i]) { $diff += $i } }
  $folder = 'D:\\\\r65-smoke\\\\4a03b65b\\\\wr1-R20261029T1633Z\\\\' + $d
  $onlyFolder = ($diff.Count -eq 1) -and ($cur[$diff[0]].Replace($folder, $tplFolder) -ceq $tpl[$diff[0]])
  Check "P8.plantilla.$d" $onlyFolder ("lineas_distintas=" + $diff.Count + " solo_dumpFolder=" + $onlyFolder)
}
$specs = [ordered]@{ 'docs/initiatives/I-65-f1-ref-cant-01.md' = '45cde765'; 'docs/automation/evidence/I-65-f1/ref-cant-01.design.json' = '2cec9656'; 'tests/RackCad.Tests/StructuralAnalysis/Fixtures/ref-cant-01.loads.json' = '3855c268'; 'docs/automation/evidence/I-65-f4/p2-R20261016T2147Z/manual-entry-sheet.json' = 'a720c0d1'; 'docs/automation/evidence/I-65-f5-prep/ov-i65-02-manual-entry-sheet.json' = '6dc0d8da'; 'docs/automation/evidence/I-65-f5-prep/ov-i65-02.loads.json' = 'dcc1a039' }
foreach ($p in $specs.Keys) {
  $b0 = (& git -C $Worktree rev-parse "4a03b65b:$p").Trim(); $bh = (& git -C $Worktree rev-parse "HEAD:$p").Trim()
  $src = $(if ($p -like '*f5-prep*') { 'bd1fcf8a' } elseif ($p -like '*p2-R2026*') { 'bd1fcf8a' } else { '51fc82f1' })
  $bs = (& git -C $Worktree rev-parse "${src}:$p").Trim()
  Check "P8.spec" ($b0.StartsWith($specs[$p]) -and $b0 -eq $bh -and $b0 -eq $bs) ("$p blob 4a03b65b=" + $b0.Substring(0, 8) + " HEAD=" + $bh.Substring(0, 8) + " $src=" + $bs.Substring(0, 8))
}

& cmd.exe /c "git -C `"$Worktree`" cat-file blob 4a03b65b:docs/automation/evidence/I-65-f5-prep/ov-i65-02-manual-entry-sheet.json > `"$blob`""
Check 'P8.hoja-OV02' ((Sha $blob).StartsWith('b15b9f9e')) ("contenido sha256=" + (Sha $blob))
Check 'P8.cargas-OV02' ((Sha 'D:\r65-smoke\4a03b65b\inputs\ov-i65-02.loads.json') -eq 'd675d9f6bb537083a63571f08fb9884c1ab085a0eca4ad1555ee98bea4f8ba4c') (Sha 'D:\r65-smoke\4a03b65b\inputs\ov-i65-02.loads.json')

$diffp = @(& git -C $Worktree diff --name-only 4a03b65b..HEAD -- src deploy assets)
Check 'P10.producto' ($diffp.Count -eq 0) ("git diff 4a03b65b..HEAD -- src deploy assets: " + $diffp.Count + " archivos")
$sw = @(& git -C $Worktree status --short); $sm = @(& git -C 'D:/Documentos/Codex/Calculadora de racks' status --short)

Check 'P10.principal' ($sm.Count -eq 0) ("status=" + $sm.Count)

$L | ForEach-Object { $_ -replace [regex]::Escape($env:USERNAME), '<usuario>' }
