# W-R2 (I-65): copia las capturas de RAM a la corrida. Las de la consola, la rejilla, la barra de estado y los avisos ya se tomaron
# recortadas a la zona tecnica. Las del dialogo «Abrir» se recortan al dialogo y se tapa su panel de navegacion (nombres de personas, de
# la organizacion y carpetas ajenas). La del modelo vacio se recorta sin la barra de tareas.
param([Parameter(Mandatory)][string]$ToolResults, [Parameter(Mandatory)][string]$Out)
$ErrorActionPreference = 'Stop'
Add-Type -AssemblyName System.Drawing
New-Item -ItemType Directory -Force -Path $Out | Out-Null
function Src([string]$suffix) { $f = @(Get-ChildItem -LiteralPath $ToolResults -Filter "mcp-computer-use-blob-*-$suffix.jpg"); if ($f.Count -ne 1) { throw "captura ${suffix}: $($f.Count)" }; $f[0].FullName }
function Copy1([string]$suffix, [string]$name) { Copy-Item -LiteralPath (Src $suffix) -Destination (Join-Path $Out $name) -Force }
# Recorta [x, y, w, h] y tapa rectangulos [x, y, w, h] (coordenadas del recorte) con gris solido
function Crop([string]$suffix, [string]$name, [int[]]$rect, [int[][]]$boxes) {
  $img = [System.Drawing.Image]::FromFile((Src $suffix))
  $bmp = New-Object System.Drawing.Bitmap $rect[2], $rect[3]
  $g = [System.Drawing.Graphics]::FromImage($bmp)
  $g.DrawImage($img, (New-Object System.Drawing.Rectangle 0, 0, $rect[2], $rect[3]), (New-Object System.Drawing.Rectangle $rect[0], $rect[1], $rect[2], $rect[3]), [System.Drawing.GraphicsUnit]::Pixel)
  $brush = New-Object System.Drawing.SolidBrush ([System.Drawing.Color]::FromArgb(128, 128, 128))
  foreach ($b in $boxes) { $g.FillRectangle($brush, $b[0], $b[1], $b[2], $b[3]) }
  $brush.Dispose(); $g.Dispose(); $img.Dispose(); $bmp.Save((Join-Path $Out $name), [System.Drawing.Imaging.ImageFormat]::Jpeg); $bmp.Dispose() }
