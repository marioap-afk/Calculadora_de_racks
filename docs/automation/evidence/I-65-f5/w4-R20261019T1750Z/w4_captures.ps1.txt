# W4 (I-65): copia las capturas de la sesion a la corrida; recorta o tapa lo identificable que no es necesario (panel lateral de los dialogos).
param([Parameter(Mandatory)][string]$ToolResults, [Parameter(Mandatory)][string]$Out)
$ErrorActionPreference = 'Stop'
Add-Type -AssemblyName System.Drawing
New-Item -ItemType Directory -Force -Path $Out | Out-Null
function Src([string]$suffix) { $f = @(Get-ChildItem -LiteralPath $ToolResults -Filter "mcp-computer-use-blob-*-$suffix.jpg"); if ($f.Count -ne 1) { throw "captura ${suffix}: $($f.Count)" }; $f[0].FullName }
function SaveJpg($bmp, [string]$name) { $bmp.Save((Join-Path $Out $name), [System.Drawing.Imaging.ImageFormat]::Jpeg) }
function Copy1([string]$suffix, [string]$name) { Copy-Item -LiteralPath (Src $suffix) -Destination (Join-Path $Out $name) -Force }
function Crop([string]$suffix, [string]$name, [int]$x, [int]$y, [int]$w, [int]$h) {
  $img = [System.Drawing.Image]::FromFile((Src $suffix)); $bmp = New-Object System.Drawing.Bitmap $w, $h
  $g = [System.Drawing.Graphics]::FromImage($bmp); $g.DrawImage($img, (New-Object System.Drawing.Rectangle 0, 0, $w, $h), (New-Object System.Drawing.Rectangle $x, $y, $w, $h), [System.Drawing.GraphicsUnit]::Pixel)
  $g.Dispose(); $img.Dispose(); SaveJpg $bmp $name; $bmp.Dispose() }
function Redact([string]$suffix, [string]$name, [int]$x, [int]$y, [int]$w, [int]$h) {
  $img = [System.Drawing.Image]::FromFile((Src $suffix)); $bmp = New-Object System.Drawing.Bitmap $img; $img.Dispose()
  $g = [System.Drawing.Graphics]::FromImage($bmp); $g.FillRectangle([System.Drawing.Brushes]::Gray, $x, $y, $w, $h); $g.Dispose()
  SaveJpg $bmp $name; $bmp.Dispose() }
