"""Barrido de secretos y datos personales en archivos a versionar. NUNCA imprime valores: solo archivo, tipo y numero de coincidencias.
Uso: python -I secrets_scan.py <archivo-o-carpeta>..."""
import os
import re
import sys

SECRETS = {
    "github-token": r"\b(ghp|gho|ghu|ghs|ghr)_[A-Za-z0-9]{30,}|github_pat_[A-Za-z0-9_]{40,}",
    "aws-key": r"\bAKIA[0-9A-Z]{16}\b",
    "private-key": r"-----BEGIN [A-Z ]*PRIVATE KEY-----",
    "slack-token": r"\bxox[abprs]-[A-Za-z0-9-]{10,}",
    "openai-anthropic-key": r"\bsk-(ant-)?[A-Za-z0-9_-]{20,}",
    "google-api-key": r"\bAIza[0-9A-Za-z_-]{35}\b",
    "jwt": r"\beyJ[A-Za-z0-9_-]{10,}\.[A-Za-z0-9_-]{10,}\.[A-Za-z0-9_-]{10,}",
    "bearer": r"(?i)\bbearer\s+[A-Za-z0-9._-]{20,}",
    "assignment": r"(?i)\b(password|passwd|pwd|secret|token|api[_-]?key|client[_-]?secret)\b\s*[:=]\s*['\"]?[^\s'\"<>]{8,}",
    "connection-string": r"(?i)(AccountKey|SharedAccessSignature|Password)=[^;\s]{8,}",
}
PERSONAL = {
    "email": r"[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}",
    "user-path": r"(?i)C:[/\\]Users[/\\](?!<usuario>)[^/\\\s<]+",
    "short-user-path": r"<forma corta local del usuario; no se versiona>",
    "names": r"<lista local de nombres y organizaciones a vigilar; no se versiona (PRIVACY HOLD)>",
}
paths = []
for a in sys.argv[1:]:
    if os.path.isdir(a):
        for d, _, fs in os.walk(a):
            paths += [os.path.join(d, f) for f in fs]
    else:
        paths.append(a)
total = 0
for p in sorted(paths):
    t = open(p, "rb").read().decode("utf-8", "replace")
    for kind, groups in (("SECRET", SECRETS), ("PERSONAL", PERSONAL)):
        for name, rx in groups.items():
            n = len(re.findall(rx, t))
            if n:
                total += n
                print("%s %s %s x%d" % (kind, name, p.replace(os.sep, "/").split("/docs/", 1)[-1], n))
print("files=%d matches=%d" % (len(paths), total))
