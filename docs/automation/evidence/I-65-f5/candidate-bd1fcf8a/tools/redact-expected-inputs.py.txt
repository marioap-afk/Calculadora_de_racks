"""Copia revisable de expected-inputs.json con el nombre del verificador redactado (PRIVACY HOLD).
Sustituye el valor de verifiedBy (y su hex UTF-8) en TODAS las cadenas, incluidas canonicalText. No toca el original.
Uso: python -I redact_expected_inputs.py <expected-inputs.json> <salida.json>"""
import hashlib
import json
import sys

src, dst = sys.argv[1:3]
raw = open(src, "rb").read()
d = json.loads(raw.decode("utf-8-sig"))
names = {r["verifiedBy"] for r in d["rows"] if r.get("verifiedBy")}
names.add(d["expected"]["verifiedBy"])
hexes = {n.encode("utf-8").hex() for n in names}
TAG = "<redactado: verificador de la tabla>"
count = {"text": 0, "hex": 0}


def walk(v):
    if isinstance(v, dict):
        return {k: walk(x) for k, x in v.items()}
    if isinstance(v, list):
        return [walk(x) for x in v]
    if isinstance(v, str):
        for n in names:
            if n in v:
                count["text"] += v.count(n)
                v = v.replace(n, TAG)
        for h in hexes:
            if h in v:
                count["hex"] += v.count(h)
                v = v.replace(h, "<redactado: hex UTF-8 del verificador>")
        return v
    return v


out = walk(d)
out["_redaction"] = {
    "originalPath": d["table"]["path"].rsplit("bin-Debug-net8.0-windows", 1)[0] + "expected-inputs.json",
    "originalSha256Bytes": hashlib.sha256(raw).hexdigest(),
    "originalBytes": len(raw),
    "redacted": "valor de verifiedBy (en rows, expected y canonicalText) y su hex UTF-8; el resto, sin cambios",
    "replacements": count,
}
text = json.dumps(out, ensure_ascii=False, indent=2) + "\n"
for n in names:
    assert n not in text
open(dst, "w", encoding="utf-8", newline="\n").write(text)
print("ok", count, hashlib.sha256(text.encode("utf-8")).hexdigest())
